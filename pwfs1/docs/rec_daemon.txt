		Gemini A&G WAVEFRONT PROCESSING SYSTEM
		======================================

		Code Executed by "Record Daemon" task
		=====================================

1) The Record Daemon task is spawned from the startup script after iocInit(),
   database initialisation with epToVxCaInitRecords(), and spawning of
   application and CAR Daemon tasks, and executes epToVxCaWriteDaemon( nProcOnBus ). nProcOnBus
   is the number of processors on the VME bus and is obtained by calling sysextNProcGet() from
   the shell.

epToVxCaWriteDaemon( nProcOnBus )
{
	Create and initialise an error context structure for this task using errorInit().

	If the number of processors, nProcOnBus, is outside the range 0-100 then
	  report an error and abort.

	For each processor, procNumber, on the bus do
	{
		Convert procNumber into a pipe name extension, pNameExtension
		Open a CA Write pipe to this processor, storing the file descriptor in
		  pRecWriteFd [procNumber], using
		  epToVxPipeOpen( fullNameProvided=FALSE, pName=CA_WRITE_PIPE_NAME,
		                  pNameExtension=pNameExtension, pipeCreate=NULL, nMsgSlots=0,
		                  maxMsgSize=0, mode=O_RDONLY, ioctlFunction=-1,
		                  timeoutPeriod=DATA_PKT_TIMEOUT_OPENPIPE,
		                  timeoutDelay=DATA_PKT_DELAY_OPENPIPE )
		{

			If fullNameProvided then
			{
				Use the pipe name provided
			}
			else
			{
				Generate the pipe name by concatenating "/pipe/" + pName + pNameExtension
			}

			If the pipeCreate function is not NULL then
			{
				Create the pipe with pipeCreate( pPipeName, nMsgSlots, maxMsgSize )
			}

			Wait until the pipe exists using
			  waitPipeExists( pPipeName, timeoutPeriod, timeoutDelay)
			{
				If the pipe doesn't already exist and a non-zero timeoutPeriod is specified
				{
					Check for the pipe's existence every timeoutDelay seconds until
					  the pipe exists or a timeout occurs.
					If a timeout occurs report an ERROR.
				}
			}

			Open the pipe in the required mode, fd = open(pPipeName, mode, 0).
			If an ioctl function was specified, execute it.
		}
		Remember the largest file descriptor found so far in widthSelect.
	} Next processor

	Repeat forever - this task never stops
	{
		Zero the file descriptor structure using FD_ZERO (& readFds).

		For each processor, procNumber do
		{
			Set the file descriptor bit corresponding to this processor
			  using FD_SET (pRecWriteFd [procNumber], & readFds)
		} Next processor

		Wait for activity on any of the file descriptors using
		  select (widthSelect, & readFds, NULL, NULL, NULL)
		  (This call will block until a message is received on onw or more pipes).

		For each processor, procNumber do
		{
			If a message has been received on the pipe from this processor,
			  FD_ISSET (pRecWriteFd [procNumber], & readFds), then
			{
				Read the message, specifying that DATA_PKT_PIPE_WRITE_MAX_BYTES be read from
				  the pRecWriteFd [procNumber] pipe and stored in pContext.pDataPacket
				If between 1 and DATA_PKT_PIPE_WRITE_MAX_BYTES have been received from the pipe
				{
					Extract the record type from the data packet using
					  (DATA_PKT_RECORD_ID (& pContext) >> 16) & 0xffff
					  which extracts the top 2 bytes from second 4 bytes of the data packet
					{
						(DATA_PKT_RECORD_ID(& pContext) translates to
						  * (uint32 *) ((& pContext)->pDataPacket + 4)
					}

					Extract the record number from the data packet using
					  DATA_PKT_RECORD_ID (& pContext) & 0xffff
					  which extracts the bottom 2 bytes from second 4 bytes of the data packet

					If the record type is SIR then
					{
						Obtain the record data type from the pSirList[] structure using
						  recordDataType = pSirList[recordNumber].type
						Obtain the record name from the pSirList[] structure using
						  strncpy( pRecordName, pSirList[recordNumber].pRecordName,
						           EPICS_MAX_BYTES_RECORD_NAME)
					}
					Else for any other type of record
					{
						Report an error - only SIR record types supported.
					}

					If the record type is supported (i.e. it is a SIR) then
					{
						If the record data type is LONG then
						{
							Extract the value from the data packet using
							  REC_FIELD_VALUE_LONG (ppCaDefTable [recordType][recordNumber]) =
							  * (long *) DATA_PKT_VALUE_PTR (& pContext)
							{
								REC_FIELD_VALUE_LONG( ppCaDefTable[recordType][recordNumber] )
								  translates to
								  * (long *)
								  (ppCaDefTable[recordType][recordNumber]->ppFieldValue[0])

								DATA_PKT_VALUE_PTR( & pContext ) translates to
								  (& pContext)->pDataPacket + 12
							}
						}
						Else if the record data type is DOUBLE then
						{
							Extract the value from the data packet using
							  REC_FIELD_VALUE_LONG (ppCaDefTable [recordType][recordNumber]) =
							  * (double *) DATA_PKT_VALUE_PTR (& pContext)
							{
								REC_FIELD_VALUE_DOUBLE( ppCaDefTable[recordType][recordNumber] )
								  translates to
								  * (double *)
								  (ppCaDefTable[recordType][recordNumber]->ppFieldValue[0])

								DATA_PKT_VALUE_PTR( & pContext ) translates to
								  (& pContext)->pDataPack
						}
						Else if the record data type is STRING then
						{
							Extract the value from the data packet using
							  strncpy( REC_FIELD_VALUE_STRING(
							           ppCaDefTable[recordType][recordNumber]),
							  DATA_PKT_VALUE_PTR( & pContext ), EPICS_MAX_BYTES_STRING_ATTRIB)
							{
								REC_FIELD_VALUE_DOUBLE( ppCaDefTable[recordType][recordNumber] )
								  translates to
								  (char *)
								  (ppCaDefTable[recordType][recordNumber]->ppFieldValue[0])

								DATA_PKT_VALUE_PTR( & pContext ) translates to
								  (& pContext)->pDataPack
						}
						Else
						{
							Undefined data type - ignore
						}

						If the record type is defined then
						{
							Write the value to the record by channel access using
							  epToVxCaWrite( ppCaDefTable[recordType][recordNumber] )
							{
								If pCaContext is NULL then report an error.

								For each field, fieldNumber, within the channel access
								  definition structure, pCaContext, up to pCaContext->nField do
								{
									If there is a valid channel access ID structure,
									  pCaContext->pChannelId[fieldNumber] then
									{
										Call ca_array_put(
										          type=pCaContext->pFieldType[fieldNumber],
										          count=1,
										          chid=pCaContext->pChannelId[fieldNumber],
										          pvalue=pCaContext->ppFieldValue[fieldNumber] )
										  to write the value for this field to the database
										  record.
									}
								} next field

								Call ca_flush_io() to flush any pending Channel Access I/O.
							}
						}
					}
				}
				Else
				{
					Report an error - an invalid number of bytes has been read.
				}
			}
		} Next processor


	} Go back and repeat forever

}
